// Remove this test-only registry adapter when the sample SDK pin includes
// temporalio/sdk-go#2914. It adds only the missing factory contract; routing and
// replay behavior are tested against the real SDK in that PR.
package internal

import (
	"fmt"

	"go.temporal.io/sdk/converter"
)

var systemNexusRegistry = make(map[SystemNexusRegistryKey]SystemNexusRegistryEntry)

type SystemNexusRegistryKey struct {
	Service   string
	Operation string
}

type SystemNexusRegistryEntry struct {
	SerializationContext func(any) converter.SerializationContext
	StorageTarget        func(any) converter.StorageDriverWorkflowInfo
}

func UpdateSystemNexusRegistry(entries map[SystemNexusRegistryKey]SystemNexusRegistryEntry) {
	for key, entry := range entries {
		if _, exists := systemNexusRegistry[key]; exists {
			panic(fmt.Sprintf("duplicate System Nexus operation %s/%s", key.Service, key.Operation))
		}
		systemNexusRegistry[key] = entry
	}
}
